const jobDB = [ { name: "Водитель автобуса", pay: 290000 }, { name: "Рыбалка", pay: 920000 }, { name: "Мусоровоз", pay: 400000 }, { name: "Кладоискатель", pay: 300000 }, { name: "Дальнобойщик", pay: 270000 }, { name: "Курьер", pay: 220000 }, { name: "Инкассатор", pay: 100000 }, { name: "Мойщик улиц", pay: 229000 }, { name: "Грибник", pay: 200000 }, { name: "Курьер (вариант 2)", pay: 130000 } ]; function renderJobs() { const input = document.getElementById("jobSearch"); const box = document.getElementById("jobResults"); const query = norm(input.value); if (!query) { box.innerHTML = "Введи название работы, чтобы увидеть заработок."; return; } const results = jobDB.filter(job => norm(job.name).includes(query) ); if (!results.length) { box.textContent = "Работа не найдена."; return; } box.innerHTML = results.map(job => `
${esc(job.name)}
Примерный заработок за час
≈ ${job.pay.toLocaleString("ru-RU")} виртов/ч
`).join(""); }
